import 'dart:convert';
import 'package:http/http.dart' as http;

class TextRazorService {
  final String apiKey = 'e5a8e159c559443a6290c0e9495efe2a22190b5be84fe9355159c139';

  Future<Map<String, dynamic>> analyzeText(String text) async {
    final url = Uri.parse('https://api.textrazor.com/');

    final response = await http.post(
      url,
      headers: {
        'x-textrazor-key': apiKey,
        'Content-Type': 'application/x-www-form-urlencoded',
      },
      body: {
        'text': text,
        'extractors': 'entities,topics,tone', // Selección de extractores
      },
    );

    if (response.statusCode == 200) {
      print('Respuesta completa: ${response.body}');  // Esto imprime toda la respuesta
      return json.decode(response.body);
    } else {
      throw Exception('Failed to load analysis');
    }
  }

}
